function auth_ensure_column(mysqli $conn, string $table, string $column, string $definitionSql): void {
    try {
        if (!preg_match('/^[a-zA-Z0-9_]+$/', $table) || !preg_match('/^[a-zA-Z0-9_]+$/', $column)) return;
        $stmt = $conn->prepare("SELECT 1 FROM information_schema.COLUMNS WHERE TABLE_SCHEMA = DATABASE() AND TABLE_NAME = ? AND COLUMN_NAME = ? LIMIT 1");
        $stmt->bind_param('ss', $table, $column); $stmt->execute(); $exists = $stmt->get_result();
        if ($exists && $exists->num_rows === 0) $conn->query("ALTER TABLE `$table` ADD COLUMN `$column` $definitionSql");
        $stmt->close();
    } catch (Throwable $e) { error_log("auth_ensure_column {$table}.{$column}: " . $e->getMessage()); }
}

function ensure_auth_schema(mysqli $conn): void {
    $conn->query("CREATE TABLE IF NOT EXISTS users (id INT AUTO_INCREMENT PRIMARY KEY, username VARCHAR(100) NOT NULL UNIQUE, password_hash VARCHAR(255) NOT NULL, role ENUM('admin','rater','user') NOT NULL DEFAULT 'user', created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP) DEFAULT CHARSET=utf8mb4");
    auth_ensure_column($conn, 'users', 'region', 'VARCHAR(100) NULL DEFAULT NULL');
    auth_ensure_column($conn, 'users', 'field_name', 'VARCHAR(100) NULL DEFAULT NULL');
    auth_ensure_column($conn, 'users', 'salary_grade', 'TINYINT UNSIGNED NULL DEFAULT NULL');
    $conn->query("CREATE TABLE IF NOT EXISTS remember_tokens (id INT AUTO_INCREMENT PRIMARY KEY, user_id INT NOT NULL, token_hash CHAR(64) NOT NULL UNIQUE, expires_at DATETIME NOT NULL, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP) DEFAULT CHARSET=utf8mb4");
    $conn->query("CREATE TABLE IF NOT EXISTS login_attempts (ip VARCHAR(45) PRIMARY KEY, attempts INT NOT NULL DEFAULT 0, locked_until DATETIME NULL, updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP) DEFAULT CHARSET=utf8mb4");
}

function issue_remember_token(mysqli $conn, int $userId, int $days = 30): string {
    $token = bin2hex(random_bytes(32)); $hash = hash('sha256', $token); $expires = date('Y-m-d H:i:s', time() + ($days * 86400));
    $stmt = $conn->prepare("INSERT INTO remember_tokens (user_id, token_hash, expires_at) VALUES (?, ?, ?)");
    $stmt->bind_param('iss', $userId, $hash, $expires); $stmt->execute(); $stmt->close(); return $token;
}

function verify_remember_token(mysqli $conn, string $token): ?array {
    $hash = hash('sha256', $token);
    $stmt = $conn->prepare("SELECT u.id, u.username, u.role FROM remember_tokens t JOIN users u ON u.id = t.user_id WHERE t.token_hash = ? AND t.expires_at > NOW() LIMIT 1");
    $stmt->bind_param('s', $hash); $stmt->execute(); $row = $stmt->get_result()->fetch_assoc(); $stmt->close(); return $row ?: null;
}

function delete_remember_token(mysqli $conn, ?string $token): void {
    if (!$token) return; $hash = hash('sha256', $token);
    $stmt = $conn->prepare("DELETE FROM remember_tokens WHERE token_hash = ?");
    $stmt->bind_param('s', $hash); $stmt->execute(); $stmt->close();
}

function purge_expired_tokens(mysqli $conn): void { $conn->query("DELETE FROM remember_tokens WHERE expires_at <= NOW()"); }

function request_ip(): string {
    foreach (['HTTP_CF_CONNECTING_IP', 'HTTP_TRUE_CLIENT_IP'] as $h) { $v = trim((string)($_SERVER[$h] ?? '')); if ($v !== '' && filter_var($v, FILTER_VALIDATE_IP)) return $v; }
    $xff = trim((string)($_SERVER['HTTP_X_FORWARDED_FOR'] ?? ''));
    if ($xff !== '') { $parts = array_filter(array_map('trim', explode(',', $xff))); if (!empty($parts)) { $last = end($parts); if (filter_var($last, FILTER_VALIDATE_IP)) return $last; } }
    $remote = $_SERVER['REMOTE_ADDR'] ?? '0.0.0.0'; return filter_var($remote, FILTER_VALIDATE_IP) ? $remote : '0.0.0.0';
}

function client_ip(): string { return request_ip(); }

function submission_rate_ok(mysqli $conn, int $limit = 500, int $windowMinutes = 60): bool {
    static $tableReady = false;
    try {
        if (!$tableReady) { $conn->query("CREATE TABLE IF NOT EXISTS submission_attempts (ip VARCHAR(45) PRIMARY KEY, `count` INT NOT NULL DEFAULT 0, window_start DATETIME NOT NULL, updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP) DEFAULT CHARSET=utf8mb4"); $tableReady = true; }
        $ip = request_ip(); $now = date('Y-m-d H:i:s'); $cut = date('Y-m-d H:i:s', time() - ($windowMinutes * 60));
        $stmt = $conn->prepare("INSERT INTO submission_attempts (ip, `count`, window_start) VALUES (?, 1, ?) ON DUPLICATE KEY UPDATE `count` = IF(window_start < ?, 1, `count` + 1), window_start = IF(window_start < ?, VALUES(window_start), window_start)");
        $stmt->bind_param('ssss', $ip, $now, $cut, $cut); $stmt->execute(); $stmt->close();
        $stmt = $conn->prepare("SELECT `count` FROM submission_attempts WHERE ip = ?"); $stmt->bind_param('s', $ip); $stmt->execute(); $row = $stmt->get_result()->fetch_assoc(); $stmt->close();
        return !$row || (int)$row['count'] <= $limit;
    } catch (Throwable $e) { error_log('submission_rate_ok failed: ' . $e->getMessage()); return true; }
}

function is_locked_out(mysqli $conn): ?int {
    $ip = client_ip(); $stmt = $conn->prepare("SELECT locked_until FROM login_attempts WHERE ip = ?"); $stmt->bind_param('s', $ip); $stmt->execute(); $row = $stmt->get_result()->fetch_assoc(); $stmt->close();
    if ($row && $row['locked_until'] && strtotime($row['locked_until']) > time()) return strtotime($row['locked_until']) - time(); return null;
}

function register_failed_attempt(mysqli $conn): void {
    $ip = client_ip(); $now = date('Y-m-d H:i:s'); $maxAttempts = 5; $lockMinutes = 15;
    $stmt = $conn->prepare("UPDATE login_attempts SET attempts = 0, locked_until = NULL WHERE ip = ? AND locked_until IS NOT NULL AND locked_until <= ?"); $stmt->bind_param('ss', $ip, $now); $stmt->execute(); $stmt->close();
    $stmt = $conn->prepare("INSERT INTO login_attempts (ip, attempts) VALUES (?, 1) ON DUPLICATE KEY UPDATE attempts = attempts + 1"); $stmt->bind_param('s', $ip); $stmt->execute(); $stmt->close();
    $stmt = $conn->prepare("UPDATE login_attempts SET locked_until = DATE_ADD(?, INTERVAL ? MINUTE) WHERE ip = ? AND attempts >= ?"); $stmt->bind_param('sisi', $now, $lockMinutes, $ip, $maxAttempts); $stmt->execute(); $stmt->close();
}

function access_field_is_valid($value): bool {
    if (!is_string($value)) return false;
    $value = trim($value);
    return $value !== ''
        && mb_strlen($value) <= 100
        && preg_match('/^[\p{L}\p{N} .&()\/-]+$/u', $value) === 1;
}

    return (is_int($value) || (is_string($value) && preg_match('/^(0|[1-9]|1[0-9]|2[0-5])$/', $value)))
        && (int)$value >= 0
        && (int)$value <= 25;

function ensure_audit_table(mysqli $conn): void {
    static $ready = false;
    if ($ready) return;

    try {
        // FIX (Bug 28): widened ENUM + auth actions
        $conn->query("CREATE TABLE IF NOT EXISTS score_audit (
            id INT AUTO_INCREMENT PRIMARY KEY,
            candidate_id INT NOT NULL,
            user_id INT NULL,
            username VARCHAR(100) NULL,
            action ENUM('submit','edit','delete','role_change','password_reset',
                        'login_failure','login_success','logout') NOT NULL,
            old_value TEXT NULL,
            new_value TEXT NULL,
            ip VARCHAR(45) NULL,
            user_agent VARCHAR(255) NULL,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            INDEX idx_candidate (candidate_id),
            INDEX idx_user (user_id),
            INDEX idx_created (created_at)
        ) DEFAULT CHARSET=utf8mb4");

        // FIX (Bug 28): if the table already existed with the old ENUM,
        // widen it once. Silently ignore if the column already matches.
        try {
            $conn->query("ALTER TABLE score_audit MODIFY COLUMN action
                ENUM('submit','edit','delete','role_change','password_reset',
                     'login_failure','login_success','logout') NOT NULL");
        } catch (Throwable $e) {
            // Already the right shape, or no ALTER permission — ignore.
        }

        $ready = true;
    } catch (Throwable $e) {
        // FIX (Bug 25): log the failure. We deliberately do NOT set
        // $ready = true, so the next call retries.
        error_log('ensure_audit_table failed: ' . $e->getMessage());
    }

// $candidateId is 0 for actions that are about an account (role change, password reset) rather than an applicant.
// $oldValue / $newValue can be arrays (stored as JSON), strings or null. Never put passwords or hashes in them.
function audit_log(mysqli $conn, int $candidateId, string $action, $oldValue = null, $newValue = null): void {
    try {
        ensure_audit_table($conn);

        $encode = function ($v) {
            if ($v === null) return null;
            return is_string($v) ? $v : json_encode($v, JSON_UNESCAPED_UNICODE);
        };
        $old = $encode($oldValue);
        $new = $encode($newValue);

        $userId   = function_exists('current_user_id') ? current_user_id() : null;
        $username = isset($_SESSION['username']) ? mb_substr((string)$_SESSION['username'], 0, 100) : null;
        $ip       = function_exists('request_ip') ? request_ip() : ($_SERVER['REMOTE_ADDR'] ?? null);
        $agent    = isset($_SERVER['HTTP_USER_AGENT']) ? mb_substr((string)$_SERVER['HTTP_USER_AGENT'], 0, 255) : null;

        // FIX: bind user_id as SQL NULL when there is no logged-in user.
        // mysqli sends SQL NULL for a PHP null when the param type is 'i'.
        $stmt = $conn->prepare("INSERT INTO score_audit
            (candidate_id, user_id, username, action, old_value, new_value, ip, user_agent)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?)");
        $stmt->bind_param('iissssss', $candidateId, $userId, $username, $action, $old, $new, $ip, $agent);
        $stmt->execute();
        $stmt->close();
    } catch (Throwable $e) {
        error_log('audit_log failed: ' . $e->getMessage());
    }

            $want = compact_position($stems[$abbr] . ' ' . $numeral); $target = null;
            foreach (leadership_level_titles() as $title => $_l) { if (compact_position_no_parens($title) === $want) { $target = $title; break; } }
            if ($target === null) continue;
            $keys = [$abbr . $numeral]; if ($numeral !== '' && isset($r2d[$numeral])) $keys[] = $abbr . $r2d[$numeral];
            foreach ($keys as $k) { if (!isset($cache['aliasToTitle'][$k])) $cache['aliasToTitle'][$k] = $target; }
        }
    }
    $raw = trim($position); if ($raw === '') return '';
    $full = compact_position($raw); if (isset($cache['titles'][$full])) return $cache['titles'][$full];
    $base = compact_position_no_parens($raw); if (isset($cache['titlesNoParens'][$base])) return $cache['titlesNoParens'][$base];
    if (isset($cache['aliasToTitle'][$full])) return $cache['aliasToTitle'][$full]; if (isset($cache['aliasToTitle'][$base])) return $cache['aliasToTitle'][$base];
    return $raw;
}

function office_has_leadership(string $office, string $position = ''): bool {
    if (in_array(strtoupper(trim($office)), offices_without_leadership(), true)) return false;
    $canonical = canonical_position_title($position);
    if (in_array(strtoupper(trim($position)), leadership_hidden_positions(), true)) return false;
    if (in_array(strtoupper(trim($canonical)), leadership_hidden_positions(), true)) return false;
    return leadership_level_for_position($canonical) !== null;
}

function record_rater_order(mysqli $conn, string $office, string $position, array $ratersInOrder): void {
    $office = strtoupper(trim($office)); $position = canonical_position_title(trim($position));
    if ($office === '' || $position === '' || empty($ratersInOrder)) return;
    $clean = [];
    foreach ($ratersInOrder as $r) { $c = canonical_rater_name((string)$r); if ($c !== '' && !in_array($c, $clean, true)) $clean[] = $c; }
    if (empty($clean)) return;
    $del = $conn->prepare("DELETE FROM position_rater_order WHERE office = ? AND position = ?"); $del->bind_param('ss', $office, $position); $del->execute(); $del->close();
    $ins = $conn->prepare("INSERT INTO position_rater_order (office, position, rater, sort_order) VALUES (?, ?, ?, ?)");
    foreach ($clean as $index => $rater) { $ins->bind_param('sssi', $office, $position, $rater, $index); $ins->execute(); }
    $ins->close();
}

function template_seed_layout(array $position, array $defaultLayout): array {
    $sections = [
        'core' => [
            'standard' => true,
            'label' => 'Core Competencies',
            'level' => 'Evidence',
            'rows' => template_seed_copy_rows($defaultLayout['sections']['core']),
        ],
        'org' => [
            'standard' => true,
            'label' => 'Organizational Competencies',
            'level' => 'Superior',
            'rows' => template_seed_copy_rows($defaultLayout['sections']['org']),
        ],
    ];

    $leadershipLevel = strtoupper(trim((string)($position['leadership_level'] ?? '')));
    if ($leadershipLevel === 'B' || $leadershipLevel === 'I') {
        $sections['lead'] = [
            'standard' => true,
            'label' => 'Leadership Competencies',
            'level' => $leadershipLevel === 'B' ? 'Basic' : 'Advanced',
            'rows' => template_seed_copy_rows($defaultLayout['sections']['lead']),
        ];
    }

    $sections['func'] = [
        'custom' => true,
        'label' => 'Functional Competencies',
        'level' => 'Superior',
        'rows' => template_seed_copy_rows($defaultLayout['sections']['func']),
    ];

    return ['sections' => $sections];
}

function template_seed_parse_functional_rows(string $text): array {
    $rows = [];
    $errors = [];
    foreach (preg_split('/\r\n|\n|\r/', $text) as $index => $line) {
        $lineNumber = $index + 1;
        if (trim($line) === '') continue;
        if (str_contains($line, "\t")) {
            $fields = explode("\t", $line, 2);
        } else {
            $fields = str_getcsv($line);
            if (count($fields) > 2) {
                $fields = [$fields[0], implode(',', array_slice($fields, 1))];
            }
        }
        if (count($fields) !== 2) {
            $errors[] = ['line' => $lineNumber, 'message' => 'Expected a title and description separated by a tab or comma.'];
            continue;
        }
        $title = trim((string)$fields[0]);
        $description = trim((string)$fields[1]);
        if ($title === '' || $description === '' || mb_strlen($title) > 255 || mb_strlen($description) > 2000) {
            $errors[] = ['line' => $lineNumber, 'message' => 'Title and description are required (maximum 255 and 2,000 characters).'];
            continue;
        }
        $rows[] = ['title' => $title, 'desc' => $description];
    }
    return [$rows, $errors];
}

function groupAvg(array $codes, array $ratings, array $extra = []): ?float {
    $vals = [];
    foreach ($codes as $c) {
        if (isset($ratings[$c]) && $ratings[$c] !== null) $vals[] = $ratings[$c];
    }
    foreach ($extra as $v) {
        if ($v !== null) $vals[] = (float)$v;
    }
    return count($vals) > 0 ? array_sum($vals) / count($vals) : null;
}

$lead_avg = $hasLeadership ? groupAvg($groups['lead'], $ratings, $customRatingsBySec['lead']) : null;
$func_avg = groupAvg($groups['func'], $ratings, $customRatingsBySec['func']);

$psycho_score    = cell_avg($ratings, core_cell_codes());
$potential_score = cell_avg($ratings, right_cell_codes($hasLeadership));

$evidenceJson = count($evidence) > 0 ? json_encode($evidence, JSON_UNESCAPED_UNICODE) : null;

$columns = ['candidate_name', 'position_title', 'office_location', 'office_name', 'interview_date', 'rater_name',
            'core_avg', 'org_avg', 'lead_avg', 'func_avg', 'evidence_json', 'psycho_ave', 'potential_ave',
            'custom_json'];
$values  = [$candidate_name, $position_title, $office_location, $office_name, $interview_date, $rater_name,
            $core_avg, $org_avg, $lead_avg, $func_avg, $evidenceJson, $psycho_score, $potential_score,
            $customJson];
$types   = 'ssssss' . 'dddd' . 's' . 'dd' . 's';

foreach ($ratings as $code => $rating) {
    $columns[] = $code;
    $values[]  = $rating;
    $types    .= 'd';
}

$placeholders = implode(',', array_fill(0, count($columns), '?'));

try {
    $conn->begin_transaction();

    $find = $conn->prepare("SELECT id FROM candidates
        WHERE candidate_name = ? AND UPPER(TRIM(office_name)) = UPPER(?) AND TRIM(position_title) = ?
        ORDER BY id DESC LIMIT 1 FOR UPDATE");
    $find->bind_param('sss', $candidate_name, $office_name, $position_title);
    $find->execute();
    $existingCandidate = $find->get_result()->fetch_assoc();
    $find->close();

    if ($existingCandidate) {
        $candidateId = (int)$existingCandidate['id'];
        $isEdit = true;

        $setParts = [];
        foreach ($columns as $col) {
            $setParts[] = "`$col` = ?";
        }
        $setSql = "UPDATE candidates SET " . implode(', ', $setParts) . " WHERE id = ?";
        $upd = $conn->prepare($setSql);
        $typesWithId = $types . 'i';
        $valuesWithId = $values;
        $valuesWithId[] = $candidateId;
        $upd->bind_param($typesWithId, ...$valuesWithId);
        $upd->execute();
        $upd->close();
    } else {
        $sql = "INSERT INTO candidates (" . implode(',', $columns) . ") VALUES ($placeholders)";
        $stmt = $conn->prepare($sql);
        $stmt->bind_param($types, ...$values);
        $stmt->execute();
        $candidateId = $conn->insert_id;
        $stmt->close();
        $isEdit = false;
    }

    $stmtScore = $conn->prepare("INSERT INTO candidate_scores (candidate_id, rater, psycho, potential, rated_by, rated_at)
        VALUES (?, ?, ?, ?, ?, NOW())
        ON DUPLICATE KEY UPDATE psycho = VALUES(psycho), potential = VALUES(potential),
            rated_by = VALUES(rated_by), rated_at = VALUES(rated_at)");
    $ratedBy = (int)current_user_id();
    foreach ($selectedRaters as $rater) {
        $stmtScore->bind_param('isddi', $candidateId, $rater, $psycho_score, $potential_score, $ratedBy);
        $stmtScore->execute();
    }
    $stmtScore->close();

    $existingRaters = position_rater_order($conn, $office_name, $position_title);
    $merged         = array_values(array_unique(array_merge($existingRaters, $selectedRaters)));
    record_rater_order($conn, $office_name, $position_title, $merged);

    $conn->commit();
    $_SESSION['last_submit'] = time();

    if ($sheet_id > 0) {
        try {
            $st = $conn->prepare("UPDATE sheets SET status = 'submitted' WHERE id = ? AND user_id = ?");
            $uid = (int)current_user_id();
            $st->bind_param('ii', $sheet_id, $uid);
            $st->execute();
            $st->close();
        } catch (Throwable $e) {
            error_log('submit_score mark submitted failed: ' . $e->getMessage());
        }
    }

    audit_log($conn, $candidateId, $isEdit ? 'edit' : 'submit',
        null,
        [
            'candidate_name' => $candidate_name,
            'position_title' => $position_title,
            'office_name'    => $office_name,
            'raters'         => $selectedRaters,
            'sheet_id'       => $sheet_id,
            'custom_rows'    => count($customRows),
        ]
    );

    respond(true, 'Saved.', 200, $wantsJson, ['candidate_id' => $candidateId]);
} catch (Throwable $e) {
    try { $conn->rollback(); } catch (Throwable $ignored) {}
    error_log('submit_score error: ' . $e->getMessage());
    respond(false, DEBUG ? $e->getMessage() : 'Could not save. Please try again.', 500, $wantsJson);
}
